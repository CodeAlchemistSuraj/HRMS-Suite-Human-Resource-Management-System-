🗄️ Database Design & Schema

Core Tables & Relationships

1. Users Table - Central Identity Store

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    username VARCHAR(100) NOT NULL UNIQUE,          -- Case-insensitive login
    email VARCHAR(150) NOT NULL UNIQUE,             -- Unique email constraint
    password_hash VARCHAR(255),                     -- Nullable for SSO users
    status user_status NOT NULL DEFAULT 'ACTIVE',   -- ACTIVE, DISABLED, LOCKED
    employee_id UUID NULL,                          -- FK to Employee module
    mfa_method mfa_method DEFAULT 'NONE',          -- TOTP, SMS, EMAIL, NONE
    mfa_secret VARCHAR(255),                       -- Encrypted TOTP secret
    failed_login_count INT DEFAULT 0,              -- Security tracking
    last_failed_login_at TIMESTAMP,                -- Security timestamp
    last_login_at TIMESTAMP,                       -- Activity tracking
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

Business Logic Implications:

  • password_hash is nullable to support SSO-only users
  • employee_id creates loose coupling with Employee module
  • failed_login_count enables account lockout after threshold
  • mfa_secret is encrypted at application level before storage

2. Roles Table - RBAC Foundation

CREATE TABLE roles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name role_name NOT NULL UNIQUE,  -- ENUM: ROLE_ADMIN, ROLE_HR, etc.
    description TEXT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

Pre-defined Roles:

  • ROLE_ADMIN - Full system access
  • ROLE_HR - Human Resources management
  • ROLE_MANAGER - Team management
  • ROLE_EMPLOYEE - Basic user access
  • ROLE_FINANCE - Financial data access
  • ROLE_AUDITOR - Read-only audit access
  • ROLE_SYSTEM - Service-to-service communication

3. User-Roles Junction Table - Many-to-Many with History

CREATE TABLE user_roles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role_id UUID NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
    assigned_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    revoked_at TIMESTAMP,                          -- Soft delete for audit
    UNIQUE(user_id, role_id)                       -- Prevent duplicate assignments
);

Audit Trail Design:

  • assigned_at tracks when role was granted
  • revoked_at enables soft deletion and audit history
  • Unique constraint prevents role duplication

4. Refresh Tokens Table - Session Management

CREATE TABLE refresh_tokens (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    token_hash VARCHAR(255) NOT NULL,              -- Hashed token value
    issued_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,                 -- Absolute expiry
    revoked_at TIMESTAMP,                          -- Manual revocation
    ip_address VARCHAR(45),                        -- IPv4/IPv6 support
    user_agent TEXT                                -- Browser/device info
);

Security Features:

  • Token Hashing: Raw tokens never stored in database
  • IP Tracking: Geographic and device session tracking
  • Cascade Delete: Automatic cleanup when user is deleted

5. Auth Audit Log Table - Compliance & Security

CREATE TABLE auth_audit_log (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    actor_user_id UUID REFERENCES users(id),       -- Null for system actions
    action VARCHAR(50) NOT NULL,                   -- LOGIN, LOGOUT, etc.
    status VARCHAR(20),                            -- SUCCESS/FAILURE
    ip_address VARCHAR(45),
    meta JSONB,                                    -- Flexible metadata storage
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

Audit Actions Tracked:

  • LOGIN, LOGIN_FAILED, LOGOUT
  • PASSWORD_RESET, TOKEN_REFRESH, TOKEN_REVOKE
  • MFA_ENABLED, MFA_DISABLED

⚙️ Service Layer Implementation

AuthService - Core Authentication Logic

Authentication Flow

public AuthResponse authenticate(AuthRequest request) {
    // 1. User Lookup
    User user = userRepository.findByUsernameIgnoreCase(request.getUsername())
        .orElseThrow(() -> new AuthenticationException("Invalid credentials"));
    
    // 2. Account Status Check
    if (user.getStatus() != UserStatus.ACTIVE) {
        auditService.recordLoginFailure(...);
        throw new AccountLockedException("Account " + user.getStatus());
    }
    
    // 3. Password Validation
    if (!passwordEncoder.matches(request.getPassword(), user.getPasswordHash())) {
        user.incrementFailedLogin();
        userRepository.save(user);
        throw new AuthenticationException("Invalid credentials");
    }
    
    // 4. MFA Validation (if enabled)
    if (user.getMfaMethod() != MfaMethod.NONE) {
        validateMfaCode(user, request.getMfaCode());
    }
    
    // 5. Success Processing
    user.resetFailedLogin();
    user.setLastLoginAt(Instant.now());
    userRepository.save(user);
    
    // 6. Token Generation
    String accessToken = tokenService.createAccessToken(...);
    RefreshTokenDto refreshToken = tokenService.createRefreshToken(...);
    
    // 7. Audit Logging
    auditService.recordLoginSuccess(...);
    
    return buildAuthResponse(accessToken, refreshToken, user);
}

Token Refresh with Rotation

public AuthResponse refreshToken(String refreshToken) {
    // 1. Token Validation
    String tokenHash = TokenHashUtil.hashToken(refreshToken);
    RefreshToken storedToken = refreshTokenRepository.findByTokenHash(tokenHash)
        .orElseThrow(() -> new TokenException("Invalid refresh token"));
    
    // 2. Security Checks
    if (storedToken.isExpired() || storedToken.isRevoked()) {
        throw new TokenException("Refresh token expired or revoked");
    }
    
    // 3. User Validation
    User user = userRepository.findById(storedToken.getUserId())
        .orElseThrow(() -> new UserNotFoundException("User not found"));
    
    // 4. Token Rotation (Security Best Practice)
    storedToken.revoke(); // Revoke old token
    refreshTokenRepository.save(storedToken);
    
    // 5. Issue New Tokens
    String newAccessToken = tokenService.createAccessToken(...);
    RefreshTokenDto newRefreshToken = tokenService.createRefreshToken(...);
    
    // 6. Audit Trail
    auditService.recordTokenRefresh(...);
    
    return buildAuthResponse(newAccessToken, newRefreshToken, user);
}

Security Benefits of Token Rotation:

  • Prevents replay attacks - old tokens become invalid
  • Shortens exposure window - compromised tokens have limited lifespan
  • Enables detection - reused tokens indicate security breach

MfaService - Multi-Factor Authentication

TOTP Provisioning Flow

public MfaProvisioningDto provisionMfa(UUID userId, MfaMethod method) {
    User user = userRepository.findById(userId)
        .orElseThrow(() -> new UserNotFoundException("User not found"));
    
    // Generate TOTP Secret
    String secret = mfaUtil.generateSecret();
    
    // Create QR Code URI
    String provisioningUri = mfaUtil.getQrCodeImageUri(
        secret, 
        user.getUsername(), 
        "HRMS-Suite"
    );
    
    // Store encrypted secret (but don't enable yet)
    user.setMfaSecret(encryptionService.encrypt(secret));
    userRepository.save(user);
    
    return MfaProvisioningDto.builder()
        .method(method)
        .provisioningUri(provisioningUri)
        .secret(secret) // One-time display for user
        .build();
}

MFA Enablement with Verification

public void enableMfa(UUID userId, MfaEnableRequest request) {
    User user = userRepository.findById(userId)
        .orElseThrow(() -> new UserNotFoundException("User not found"));
    
    // Verify the provided code matches expected
    String decryptedSecret = encryptionService.decrypt(user.getMfaSecret());
    if (!mfaUtil.verifyCode(request.getVerificationCode(), decryptedSecret)) {
        throw new MfaValidationException("Invalid verification code");
    }
    
    // Only enable after successful verification
    user.setMfaMethod(request.getMethod());
    userRepository.save(user);
}

TOTP Security Implementation:

  • RFC 6238 compliant - Standard time-based algorithm
  • 30-second windows - Balance security and usability
  • QR code generation - Easy mobile app integration
  • Secret encryption - Database-level protection

🔐 Security Infrastructure

JWT Token Provider

Token Generation

public String generateToken(String subject, Instant expiration, String jti) {
    return Jwts.builder()
        .setSubject(subject)                    // User ID
        .setIssuer(jwtConfig.getIssuer())       // HRMS-Suite
        .setIssuedAt(Date.from(Instant.now()))  // Issued time
        .setExpiration(Date.from(expiration))   // Expiry time
        .setId(jti)                             // Unique token ID
        .claim("preferred_username", username)  // Human-readable ID
        .claim("roles", roles)                  // Authorization claims
        .claim("employeeId", employeeId)        // Cross-module reference
        .signWith(getSecretKey())               // HMAC-SHA signing
        .compact();
}

JWT Authentication Filter

Request Processing Flow

protected void doFilterInternal(HttpServletRequest request, 
                              HttpServletResponse response, 
                              FilterChain filterChain) {
    
    // 1. Extract Authorization Header
    String authHeader = request.getHeader("Authorization");
    if (authHeader == null || !authHeader.startsWith("Bearer ")) {
        filterChain.doFilter(request, response);
        return;
    }
    
    // 2. Extract and Validate Token
    String token = authHeader.substring(7);
    if (jwtTokenProvider.validateToken(token)) {
        
        // 3. Parse Claims and Create Authentication
        TokenClaims claims = tokenService.parseAndValidateAccessToken(token);
        List<SimpleGrantedAuthority> authorities = extractAuthorities(claims);
        
        // 4. Set Security Context
        UsernamePasswordAuthenticationToken authentication = 
            new UsernamePasswordAuthenticationToken(claims.getSub(), null, authorities);
        authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
        
        SecurityContextHolder.getContext().setAuthentication(authentication);
    }
    
    filterChain.doFilter(request, response);
}

Security Context Propagation:

  • Thread-local storage of authentication details
  • Automatic cleanup after request processing
  • Method-level security integration via @PreAuthorize

Password Security

Argon2 Password Encoding

@Component
public class PasswordEncoderUtil {
    private final PasswordEncoder passwordEncoder;
    
    public PasswordEncoderUtil() {
        this.passwordEncoder = Argon2PasswordEncoder.defaultsForSpringSecurity_v5_8();
    }
    
    public String encode(String rawPassword) {
        return passwordEncoder.encode(rawPassword);
    }
    
    public boolean matches(String rawPassword, String encodedPassword) {
        return passwordEncoder.matches(rawPassword, encodedPassword);
    }
}

Argon2 Advantages:

  • Time and memory configurable - Adaptive to hardware
  • Salt automatically handled - No manual salt management

JWT Utility - Token Management

Claims Extraction & Validation

@Component
public class JwtUtil {
    
    public TokenClaims extractTokenClaims(Claims claims) {
        TokenClaims tokenClaims = new TokenClaims();
        
        // Extract standard claims
        tokenClaims.setSub(UUID.fromString(claims.getSubject()));
        tokenClaims.setPreferredUsername(claims.get("preferred_username", String.class));
        
        // Extract custom claims
        List<String> rolesList = claims.get("roles", List.class);
        if (rolesList != null) {
            List<RoleName> roles = rolesList.stream()
                .map(role -> RoleName.valueOf(role))
                .collect(Collectors.toList());
            tokenClaims.setRoles(roles);
        }
        
        // Extract timestamps
        tokenClaims.setIat(claims.getIssuedAt().toInstant());
        tokenClaims.setExp(claims.getExpiration().toInstant());
        tokenClaims.setJti(claims.getId());
        
        return tokenClaims;
    }
}

Claims Processing Logic:

  • Type safety: UUID conversion with validation
  • Null safety: Safe extraction of optional claims
  • Enum mapping: String to RoleName enum conversion

Token Hash Utility - Secure Storage

Cryptographic Hashing

@UtilityClass
public class TokenHashUtil {
    
    public static String hashToken(String token) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(token.getBytes(StandardCharsets.UTF_8));
            return Base64.getEncoder().encodeToString(hash);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("SHA-256 algorithm not available", e);
        }
    }
    
    public static boolean verifyToken(String token, String hashedToken) {
        return hashToken(token).equals(hashedToken);
    }
}

Security Design:

  • SHA-256: Cryptographically secure hashing algorithm
  • Base64 encoding: Database-friendly string representation
  • Deterministic: Same input always produces same output for verification

MFA Utility - TOTP Implementation

TOTP Code Generation & Verification

@Component
@Slf4j
public class MfaUtil {
    
    public String generateSecret() {
        return new DefaultSecretGenerator().generate();  // 32-character base32
    }
    
    public String getQrCodeImageUri(String secret, String username, String issuer) {
        QrData data = new QrData.Builder()
            .label(username)
            .secret(secret)
            .issuer(issuer)
            .algorithm(HashingAlgorithm.SHA1)
            .digits(6)
            .period(30)  // 30-second windows
            .build();
        
        QrGenerator generator = new ZxingPngQrGenerator();
        byte[] imageData = generator.generate(data);
        
        return getDataUriForImage(imageData, generator.getImageMimeType());
    }
    
    public boolean verifyCode(String code, String secret) {
        TimeProvider timeProvider = new SystemTimeProvider();
        CodeGenerator codeGenerator = new DefaultCodeGenerator();
        CodeVerifier verifier = new DefaultCodeVerifier(codeGenerator, timeProvider);
        return verifier.isValidCode(secret, code);
    }
}

TOTP Configuration:

  • Algorithm: SHA-1 (compatible with most authenticator apps)
  • Digits: 6-digit codes (industry standard)
  • Period: 30-second intervals (balance security and usability)
  • Window: ±1 time period for clock drift tolerance

Security Configuration

Comprehensive Security Setup

@Configuration
@EnableWebSecurity
@EnableMethodSecurity(prePostEnabled = true)
public class SecurityConfig {
    
    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .cors(cors -> cors.configurationSource(corsConfigurationSource()))
            .csrf(AbstractHttpConfigurer::disable)  // Stateless API - JWT protected
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(authz -> authz
                .requestMatchers("/api/v1/auth/login").permitAll()
                .requestMatchers("/api/v1/auth/refresh").permitAll()
                .requestMatchers("/api/v1/auth/password-reset/**").permitAll()
                .requestMatchers("/api/v1/auth/introspect").hasRole("SYSTEM")
                .requestMatchers("/api/v1/auth/revoke").hasAnyRole("ADMIN", "SYSTEM")
                .requestMatchers("/api/v1/users/**").authenticated()
                .requestMatchers("/api/v1/roles/**").hasRole("ADMIN")
                .anyRequest().authenticated()
            )
            .addFilterBefore(jwtAuthenticationFilter, UsernamePasswordAuthenticationFilter.class);
        
        return http.build();
    }
    
    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuration = new CorsConfiguration();
        configuration.setAllowedOriginPatterns(Arrays.asList("*"));  // Configure appropriately
        configuration.setAllowedMethods(Arrays.asList("GET", "POST", "PUT", "DELETE", "OPTIONS"));
        configuration.setAllowedHeaders(Arrays.asList("*"));
        configuration.setAllowCredentials(true);
        
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", configuration);
        return source;
    }
}

CORS Security Considerations:

  • Origin patterns: Restrict to known domains in production
  • Credentials: Allow cookies/auth headers
  • Methods: Limit to required HTTP methods
